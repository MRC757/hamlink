cmake_minimum_required(VERSION 3.10)
project(hamlink_core C)

set(CMAKE_C_STANDARD 99)
add_library(hamlink_core STATIC hdlc.c afsk.c fx25.c mesh.c tone.c tnc.c)
target_include_directories(hamlink_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(hamlink_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_library(M_LIB m)
if(M_LIB)
  target_link_libraries(hamlink_core PUBLIC ${M_LIB})
endif()
target_compile_options(hamlink_core PRIVATE -Wall -Wextra -O2)

# Shared library used by the Python test harness (host only).
if(NOT ANDROID)
  add_library(hamlink_core_shared SHARED hdlc.c afsk.c fx25.c mesh.c tone.c tnc.c)
  target_include_directories(hamlink_core_shared PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  if(M_LIB)
    target_link_libraries(hamlink_core_shared PUBLIC ${M_LIB})
  endif()
  target_compile_options(hamlink_core_shared PRIVATE -Wall -Wextra -O2)
endif()

# Voice clips (needs the vendored Codec2 target from the top-level project).
if(TARGET codec2_lite)
  add_library(hamlink_voice STATIC voice.c)
  target_include_directories(hamlink_voice PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(hamlink_voice PUBLIC codec2_lite)
  set_target_properties(hamlink_voice PROPERTIES POSITION_INDEPENDENT_CODE ON)
  # COFDMTV modem (vendored Rattlegram code, header-only templates) behind a C wrapper.
  add_library(hamlink_ofdm STATIC ofdm.cpp)
  target_include_directories(hamlink_ofdm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
                                           PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/aicodix)
  target_compile_features(hamlink_ofdm PRIVATE cxx_std_17)
  target_compile_options(hamlink_ofdm PRIVATE -O2 -w)
  set_target_properties(hamlink_ofdm PROPERTIES POSITION_INDEPENDENT_CODE ON)
  # MT63-2000 modem (vendored fldigi MT63 code) behind a C wrapper.
  set(MT63_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../third_party/mt63)
  add_library(hamlink_mt63 STATIC mt63.cpp ${MT63_DIR}/dsp.cxx ${MT63_DIR}/mt63base.cxx)
  target_include_directories(hamlink_mt63 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${MT63_DIR})
  target_compile_features(hamlink_mt63 PRIVATE cxx_std_17)
  target_compile_options(hamlink_mt63 PRIVATE -O2 -w)
  set_target_properties(hamlink_mt63 PROPERTIES POSITION_INDEPENDENT_CODE ON)

  # Flat C API (hamlink_api.h): used by the Windows app (P/Invoke), the Android JNI layer and the tests.
  set(HL_API_SRCS hamlink_api.c audiocheck.c voice.c hdlc.c afsk.c fx25.c psk.c mesh.c tone.c tnc.c geo.c nmea.c report.c mapmodel.c schemas_gen.c)
  add_library(hamlink_api_static STATIC ${HL_API_SRCS})
  target_include_directories(hamlink_api_static PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(hamlink_api_static PUBLIC hamlink_ofdm hamlink_mt63 codec2_lite)
  set_target_properties(hamlink_api_static PROPERTIES POSITION_INDEPENDENT_CODE ON LINKER_LANGUAGE CXX)
  target_compile_options(hamlink_api_static PRIVATE -Wall -Wextra -O2)
  add_library(hamlink_api SHARED ${HL_API_SRCS})
  target_include_directories(hamlink_api PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(hamlink_api PUBLIC hamlink_ofdm hamlink_mt63 codec2_lite)
  set_target_properties(hamlink_api PROPERTIES LINKER_LANGUAGE CXX)
  target_compile_options(hamlink_api PRIVATE -Wall -Wextra -O2 -fvisibility=hidden)
  if(NOT ANDROID)
    add_library(hamlink_core_shared_voice SHARED voice.c hdlc.c afsk.c fx25.c mesh.c tone.c tnc.c)
    target_include_directories(hamlink_core_shared_voice PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(hamlink_core_shared_voice PUBLIC codec2_lite)
  endif()
endif()
